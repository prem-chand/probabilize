def numeric_grad(f, arrays, i, h=1e-6):
    x = arrays[i]
    grad = np.zeros_like(x)
    for idx in np.ndindex(x.shape):
        old = x[idx]
        x[idx] = old + h
        hi = f(*arrays)
        x[idx] = old - h
        lo = f(*arrays)
        x[idx] = old
        grad[idx] = (hi - lo) / (2 * h)
    return grad

def unbroadcast(grad, shape):
    """Sum `grad` down to `shape`, undoing NumPy broadcasting."""
    while grad.ndim > len(shape):
        grad = grad.sum(axis=0)
    for axis, n in enumerate(shape):
        if n == 1 and grad.shape[axis] != 1:
            grad = grad.sum(axis=axis, keepdims=True)
    return grad


def _axes(axis, ndim):
    if axis is None:

    axes = axis if isinstance(axis, tuple) else (axis,)
    return tuple(a % ndim for a in axes)


def _matmul_backward(g, a, b):
    a2 = a[None, :] if a.ndim == 1 else a
    b2 = b[:, None] if b.ndim == 1 else b
    if a.ndim == 1:
        g = np.expand_dims(g, -2)
    if b.ndim == 1:
        g = np.expand_dims(g, -1)
    ga = g @ np.swapaxes(b2, -1, -2)
    gb = np.swapaxes(a2, -1, -2) @ g
    if a.ndim == 1:
        ga = ga.squeeze(-2)
    if b.ndim == 1:
        gb = gb.squeeze(-1)
    return ga, gb

    def __init__(self, data):

        self.grad = None
        self._parents = ()
        self._backward = None

    @property
    def shape(self):
        return self.data.shape

    @property
    def ndim(self):
        return self.data.ndim

    def __repr__(self):
        return f"Tensor({self.data!r})"

    @staticmethod
    def op(forward, backward, *parents):
        parents = tuple(p if isinstance(p, Tensor) else Tensor(p) for p in parents)
        out = Tensor(forward(*(p.data for p in parents)))
        out._parents = parents
        out._backward = backward
        return out

    def backward(self, grad=None):
        order, seen, stack = [], set(), [(self, False)]
        while stack:
            node, expanded = stack.pop()
            if expanded:
                order.append(node)
                continue
            if id(node) in seen:
                continue
            seen.add(id(node))
            stack.append((node, True))
            stack.extend((p, False) for p in node._parents)
        self.grad = np.ones_like(self.data) if grad is None else np.asarray(grad, dtype=np.float64)
        for node in reversed(order):
            if node._backward is None or node.grad is None:
                continue
            grads = node._backward(node.grad, *(p.data for p in node._parents))
            for parent, g in zip(node._parents, grads):
                g = unbroadcast(np.asarray(g, dtype=np.float64), parent.shape)
                parent.grad = g if parent.grad is None else parent.grad + g

def numeric_grad(f, arrays, i, h=1e-6):
    x = arrays[i]
    grad = np.zeros_like(x)
    for idx in np.ndindex(x.shape):
        old = x[idx]
        x[idx] = old + h
        hi = f(*arrays)
        x[idx] = old - h
        lo = f(*arrays)
        x[idx] = old
        grad[idx] = (hi - lo) / (2 * h)
    return grad


def gradcheck(name, build, *shapes, positive=False):
    arrays = [rng.normal(size=s) for s in shapes]
