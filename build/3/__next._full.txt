import numpy as np

from mine.autograd import Tensor


class Linear:
    def __init__(self, n_in, n_out, seed=0):
        rng = np.random.default_rng(seed)
        bound = 1.0 / np.sqrt(n_in)
        self.weight = Tensor(rng.uniform(-bound, bound, size=(n_in, n_out)))
        self.bias = Tensor(np.zeros(n_out))

    def __call__(self, x):
        return x @ self.weight + self.bias

    def parameters(self):
        return [self.weight, self.bias]


class MLP:
    def __init__(self, sizes, activation="relu", seed=0):
        if activation not in ("relu", "tanh"):
            raise ValueError(f"activation must be 'relu' or 'tanh', got {activation!r}")
        self.activation = activation
        self.layers = [Linear(a, b, seed=seed + i) for i, (a, b) in enumerate(zip(sizes, sizes[1:]))]

    def __call__(self, x):

import numpy as np


def unbroadcast(grad, shape):
    """Sum `grad` down to `shape`, undoing NumPy broadcasting."""
    while grad.ndim > len(shape):
        grad = grad.sum(axis=0)
    for axis, n in enumerate(shape):
        if n == 1 and grad.shape[axis] != 1:
            grad = grad.sum(axis=axis, keepdims=True)
    return grad


def _axes(axis, ndim):
    if axis is None:
        return tuple(range(ndim))
    axes = axis if isinstance(axis, tuple) else (axis,)
    return tuple(a % ndim for a in axes)


def _matmul_backward(g, a, b):
    a2 = a[None, :] if a.ndim == 1 else a
    b2 = b[:, None] if b.ndim == 1 else b
    if a.ndim == 1:

    if b.ndim == 1:
        g = np.expand_dims(g, -1)

        ga = ga.squeeze(-2)
    if b.ndim == 1:
        gb = gb.squeeze(-1)
    return ga, gb

    def op(forward, backward, *parents):

        out = Tensor(forward(*(p.data for p in parents)))
        out._parents = parents
        out._backward = backward
        return out

    def backward(self, grad=None):

        while stack:
            node, expanded = stack.pop()
            if expanded:
                order.append(node)
                continue
            if id(node) in seen:
                continue

            stack.append((node, True))
