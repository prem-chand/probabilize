class Linear:
    def __init__(self, n_in, n_out, seed=0):
        rng = np.random.default_rng(seed)
        bound = 1.0 / np.sqrt(n_in)
        self.weight = Tensor(rng.uniform(-bound, bound, size=(n_in, n_out)))
        self.bias = Tensor(np.zeros(n_out))

    def __call__(self, x):
        return x @ self.weight + self.bias

    def parameters(self):
        return [self.weight, self.bias]


class MLP:
    def __init__(self, sizes, activation="relu", seed=0):
        if activation not in ("relu", "tanh"):
            raise ValueError(f"activation must be 'relu' or 'tanh', got {activation!r}")
        self.activation = activation
        self.layers = [Linear(a, b, seed=seed + i) for i, (a, b) in enumerate(zip(sizes, sizes[1:]))]

    def __call__(self, x):
        if not isinstance(x, Tensor):
            x = Tensor(x)
        for i, layer in enumerate(self.layers):
            x = layer(x)
            if i < len(self.layers) - 1:
                x = x.relu() if self.activation == "relu" else x.tanh()
        return x

    def parameters(self):
        return [p for layer in self.layers for p in layer.parameters()]


class SGD:
    def __init__(self, params, lr=0.01):
        self.params = list(params)
        self.lr = lr

    def zero_grad(self):
        for p in self.params:
            p.grad = None

    def step(self):
        for p in self.params:
            if p.grad is not None:
                p.data -= self.lr * p.grad

    """Sum `grad` down to `shape`, undoing NumPy broadcasting."""
    while grad.ndim > len(shape):

def forward_kinematics(joints):
    """Elbow and hand positions for joint angles of shape (..., 2)."""
    q = np.asarray(joints, dtype=np.float64)
    q1, q12 = q[..., 0], q[..., 0] + q[..., 1]
    elbow = np.stack([LINK * np.cos(q1), LINK * np.sin(q1)], axis=-1)
    hand = elbow + np.stack([LINK * np.cos(q12), LINK * np.sin(q12)], axis=-1)
    return elbow, hand


def _segment_distance(px, py, a, b):
    ab = b - a
    t = ((px - a[0]) * ab[0] + (py - a[1]) * ab[1]) / max(float(ab @ ab), 1e-12)
    t = np.clip(t, 0.0, 1.0)
    return np.hypot(px - (a[0] + t * ab[0]), py - (a[1] + t * ab[1]))
