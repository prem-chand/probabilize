This file is the only physics in the Build. The widget draws what it is sent
and never computes geometry of its own, so Python and the page cannot disagree.

import numpy as np

LINK = 1.0
MAX_DELTA = 0.2
REACH = 0.15
BLOCK_HALF = 0.25
BLOCKS = ("red", "green", "blue")
START_JOINTS = (np.pi / 2, -np.pi / 2)

# The camera: a square window on the table, centred above the base so every
# block and the reachable upper half of the workspace stay in frame.
VIEW_CENTER = (0.0, 0.8)
VIEW_HALF = 2.0

GROUND, LINK_LABEL, JOINT, HAND, RED, GREEN, BLUE, MARKER, TARGET = range(9)
BLOCK_LABELS = {"red": RED, "green": GREEN, "blue": BLUE}

PALETTE = np.array(
    [
        [24, 24, 24],  # ground
        [110, 110, 110],  # arm link
        [170, 170, 170],  # joint
        [240, 240, 240],  # hand
        [214, 48, 49],  # red block
        [46, 160, 67],  # green block
        [40, 90, 220],  # blue block
        [240, 170, 0],  # marker
        [80, 80, 80],  # target outline
    ],
    dtype=np.uint8,
)


def forward_kinematics(joints):
    """Elbow and hand positions for joint angles of shape (..., 2)."""
    q = np.asarray(joints, dtype=np.float64)
    q1, q12 = q[..., 0], q[..., 0] + q[..., 1]
    elbow = np.stack([LINK * np.cos(q1), LINK * np.sin(q1)], axis=-1)
    hand = elbow + np.stack([LINK * np.cos(q12), LINK * np.sin(q12)], axis=-1)
    return elbow, hand


def _segment_distance(px, py, a, b):
    ab = b - a
    t = ((px - a[0]) * ab[0] + (py - a[1]) * ab[1]) / max(float(ab @ ab), 1e-12)
    t = np.clip(t, 0.0, 1.0)
    return np.hypot(px - (a[0] + t * ab[0]), py - (a[1] + t * ab[1]))


_GRIDS = {}


def _grid(size):
    if size not in _GRIDS:
        pixel = 2 * VIEW_HALF / size
        centers = (np.arange(size) + 0.5) * pixel
        xs = VIEW_CENTER[0] - VIEW_HALF + centers
        ys = VIEW_CENTER[1] + VIEW_HALF - centers
        _GRIDS[size] = (*np.meshgrid(xs, ys), pixel)
    return _GRIDS[size]


class Blockworld:
    def __init__(self, seed=0, target=None):
        self._target = target
        self.reset(seed)

    def reset(self, seed=0):
        rng = np.random.default_rng(seed)
        self.seed = seed
        self._joints = np.array(START_JOINTS, dtype=np.float64)
        elbow, hand = forward_kinematics(self._joints)
        base = np.zeros(2)
        placed = []
        while len(placed) < len(BLOCKS):
            radius = rng.uniform(0.7, 1.8, 64)
            angle = rng.uniform(np.radians(15), np.radians(165), 64)
            px, py = radius * np.cos(angle), radius * np.sin(angle)
            clear = np.minimum(
                _segment_distance(px, py, base, elbow), _segment_distance(px, py, elbow, hand)
            ) > 0.5
            for i in np.flatnonzero(clear):
                p = np.array([px[i], py[i]])
                if len(placed) < len(BLOCKS) and all(np.hypot(*(p - q)) > 0.75 for q in placed):
                    placed.append(p)
        self.blocks = dict(zip(BLOCKS, placed))
        self.target = self._target if self._target is not None else BLOCKS[int(rng.integers(3))]
        if self.target not in BLOCKS:
            raise ValueError(f"target must be one of {BLOCKS}, got {self.target!r}")
        self.steps = 0
        return self.state()

    def joints(self):
        return self._joints.copy()

    def hand(self):
        return forward_kinematics(self._joints)[1]

    def distance(self):
        return float(np.linalg.norm(self.hand() - self.blocks[self.target]))

    def reached(self):
        return self.distance() <= REACH

    def state(self):
        """Joint angles, then red, green and blue block positions: shape (8,)."""
        return np.concatenate([self._joints, *(self.blocks[name] for name in BLOCKS)])

    def step(self, action):
        delta = np.clip(np.asarray(action, dtype=np.float64).reshape(2), -MAX_DELTA, MAX_DELTA)
        self._joints = (self._joints + delta + np.pi) % (2 * np.pi) - np.pi
        self.steps += 1
        distance = self.distance()
        done = distance <= REACH
        return self.state(), -distance, done, {"distance": distance, "reached": done}

    def labels(self, size, markers=()):
        x, y, pixel = _grid(size)
        out = np.zeros((size, size), dtype=np.uint8)
        target = self.blocks[self.target]
        ring = BLOCK_HALF + max(0.06, pixel)
        near = np.maximum(np.abs(x - target[0]), np.abs(y - target[1]))
        out[near <= ring] = TARGET
        for name in BLOCKS:
            bx, by = self.blocks[name]
            out[(np.abs(x - bx) <= BLOCK_HALF) & (np.abs(y - by) <= BLOCK_HALF)] = BLOCK_LABELS[name]
        base = np.zeros(2)
        elbow, hand = forward_kinematics(self._joints)
        width = max(0.05, 0.5 * pixel)
        for a, b in ((base, elbow), (elbow, hand)):
            out[_segment_distance(x, y, a, b) <= width] = LINK_LABEL
        joint_radius = max(0.09, 0.6 * pixel)
        for c in (base, elbow):
            out[np.hypot(x - c[0], y - c[1]) <= joint_radius] = JOINT
        out[np.hypot(x - hand[0], y - hand[1]) <= joint_radius] = HAND
        marker_radius = max(0.07, 0.6 * pixel)
        for m in markers:
            mx, my = np.asarray(m, dtype=np.float64).reshape(2)
            out[np.hypot(x - mx, y - my) <= marker_radius] = MARKER
        return out

    def rgb(self, size):
        return PALETTE[self.labels(size)]

    def summary(self):
        r = lambda v: [round(float(c), 3) for c in v]
        return {
            "target": self.target,
            "hand": r(self.hand()),
            "joints": r(np.degrees(self._joints)),
            "blocks": {name: r(self.blocks[name]) for name in BLOCKS},
            "distance": round(self.distance(), 3),
            "reached": self.reached(),
        }

class CheckFailed(Exception):
    def __init__(self, message, file=None, symbol=None):
        super().__init__(message)
        self.message = message
        self.file = file
        self.symbol = symbol


def expect(condition, message, file=None):
    if not condition:
        raise CheckFailed(message, file=file)


def _module_file(module):
    return module.replace(".", "/") + ".py"


def require(module, *symbols):
    """Import a learner module and fail naming the file and first missing symbol."""
    file = _module_file(module)
    try:
        loaded = importlib.import_module(module)
    except ModuleNotFoundError as error:
        if error.name is not None and (module == error.name or module.startswith(error.name + ".")):
            raise CheckFailed(f"{file} is missing from your workspace.", file=file) from None
        raise
    for symbol in symbols:
        if not hasattr(loaded, symbol):
            raise CheckFailed(f"{file} does not define {symbol}.", file=file, symbol=symbol)
    return loaded


def _learner_file(filename):
    """`mine/nn.py` for a frame inside a learner file, else None."""
    parts = os.path.normpath(filename).split(os.sep)
    for i, part in enumerate(parts[:-1]):
        if part in LEARNER_ROOTS:
            return "/".join(parts[i:])
    return None


def _innermost_learner_frame(tb):
    found = None
    for frame, _ in traceback.walk_tb(tb):
        file = _learner_file(frame.f_code.co_filename)
        if file is not None:
            name = getattr(frame.f_code, "co_qualname", frame.f_code.co_name)
            found = (file, None if name == "<module>" else name)
    return found

_NO_MODULE = re.compile(r"No module named '([\w.]+)'")


def _result(ok, message, file=None, symbol=None, trace=None):
    return {"ok": ok, "message": message, "file": file, "symbol": symbol, "traceback": trace}


def run_check(source):
    """Run a check's source in a fresh namespace and describe the outcome."""
    try:
        exec(compile(source, "<check>", "exec"), {"__name__": "__check__"})
    except CheckFailed as failure:

    except ImportError as error:
        text = str(error)
        match = _CANNOT_IMPORT.search(text)
        if match and match.group(2).split(".")[0] in LEARNER_ROOTS:
            file = _module_file(match.group(2))
            return _result(False, f"{file} does not define {match.group(1)}.", file, match.group(1), traceback.format_exc())
        match = _NO_MODULE.search(text)
        if match and match.group(1).split(".")[0] in LEARNER_ROOTS:
            file = _module_file(match.group(1))
            return _result(False, f"{file} is missing from your workspace.", file, None, traceback.format_exc())

    except NotImplementedError as error:
        where = _innermost_learner_frame(error.__traceback__)
        if where is None:
            return _result(False, f"NotImplementedError: {error}", None, None, traceback.format_exc())
        file, symbol = where
        detail = f": {error}" if str(error) else ""
        label = f"{symbol} in {file}" if symbol else file

    except Exception as error:

        file, symbol = where if where else (None, None)

    return _result(True, "Check passed.")

def numeric_grad(f, arrays, i, h=1e-6):
    x = arrays[i]
    grad = np.zeros_like(x)
    for idx in np.ndindex(x.shape):
        old = x[idx]
        x[idx] = old + h
        hi = f(*arrays)
        x[idx] = old - h
        lo = f(*arrays)
        x[idx] = old
        grad[idx] = (hi - lo) / (2 * h)
    return grad
