def run_gd(eta, x0, K):
    x = x0
    vals = [a * x**2]
    xs = [x]
    for _ in range(K):
        g = 2 * a * x
        x = x - eta * g
        xs.append(x)
        vals.append(a * x**2)
    return np.array(xs), np.array(vals)
