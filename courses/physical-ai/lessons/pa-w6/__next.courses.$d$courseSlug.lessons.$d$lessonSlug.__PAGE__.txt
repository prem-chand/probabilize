# True alignment: text lives in LLM dim via unknown linear map + noise
W_true = np.random.randn(D_v, D_llm) * 0.5
t = v @ W_true + 0.05 * np.random.randn(N, D_llm)

def norm(x):
    return x / (np.linalg.norm(x, axis=1, keepdims=True) + 1e-8)

def retrieval_acc(v_proj, t_emb):
    s = norm(v_proj) @ norm(t_emb).T
    return float(np.mean(s.argmax(axis=1) == np.arange(len(v_proj))))

# --- Stage 1: projector-only (freeze t as LLM space) ---
W_proj, _, _, _ = np.linalg.lstsq(v, t, rcond=None)
acc1 = retrieval_acc(v @ W_proj, t)
print(f"Stage 1 projector-only retrieval acc: {acc1:.2f}")

# --- Naive: random projector (no alignment stage) ---
W_rand = np.random.randn(D_v, D_llm) * 0.1
acc0 = retrieval_acc(v @ W_rand, t)
print(f"Random projector acc:                 {acc0:.2f}")

# Fake instructions: map (image, instruction_id) → answer embedding
instr = np.random.randn(N, D_llm) * 0.2
